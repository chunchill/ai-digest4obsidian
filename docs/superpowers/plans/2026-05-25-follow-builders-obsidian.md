# Follow Builders Obsidian Plugin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a TypeScript Obsidian plugin that manually syncs raw Follow Builders central feed items into date-based Markdown files in a configured vault folder.

**Architecture:** Keep Obsidian integration thin and move feed parsing, slugging, Markdown rendering, and sync orchestration into testable modules. The plugin fetches public JSON endpoints, normalizes feed items, deduplicates by stable IDs, and writes one Markdown file per item under `Follow Builders/YYYY-MM-DD/`.

**Tech Stack:** Obsidian Plugin API, TypeScript, esbuild, Vitest, Node npm scripts.

```ts
export function safeFileName(input: string, maxLength = 80): string {
  const normalized = input
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLowerCase()
    .replace(/['"]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, maxLength)
    .replace(/-+$/g, "");

  return normalized || "untitled";
}

export function dateFolderFromIso(input: string, fallback = new Date()): string {
  const parsed = new Date(input);
  const date = Number.isNaN(parsed.getTime()) ? fallback : parsed;
  return date.toISOString().slice(0, 10);
}

export function truncateText(input: string, maxLength: number): string {
  return input.trim().slice(0, maxLength).trim();
}
```

```ts
import type { FeedItem } from "./types";

function yamlString(value: string): string {
  return `"${value.replace(/\\/g, "\\\\").replace(/"/g, '\\"')}"`;
}

function labelForMetadataKey(key: string): string {
  return key
    .replace(/([A-Z])/g, " $1")
    .replace(/^./, (char) => char.toUpperCase());
}

export function renderFeedItemMarkdown(item: FeedItem, syncedAt: string): string {
  const optionalAuthor = item.author ? `author: ${yamlString(item.author)}\n` : "";
  const optionalHandle = item.handle ? `handle: ${yamlString(item.handle)}\n` : "";

  const metadataLines = Object.entries(item.metadata)
    .filter(([, value]) => value !== undefined && value !== null && value !== "")
    .map(([key, value]) => `- ${labelForMetadataKey(key)}: ${String(value)}`);

  const bioSection = item.bio ? `\n## Bio\n\n${item.bio}\n` : "";
  const metadataSection = metadataLines.length > 0 ? `\n## Metadata\n\n${metadataLines.join("\n")}\n` : "";

  return `---\nsource: ${item.source}\ntitle: ${yamlString(item.title)}\n${optionalAuthor}${optionalHandle}url: ${yamlString(item.url)}\ncreated: ${yamlString(item.createdAt)}\nsynced: ${yamlString(syncedAt)}\nfollow_builders_id: ${yamlString(item.id)}\ntags:\n  - follow-builders\n  - ai\n---\n\n# ${item.title}\n\n> Source: [Original link](${item.url})\n${bioSection}\n## Content\n\n${item.body}\n${metadataSection}`;
}
```

```ts
import { requestUrl } from "obsidian";
import { truncateText } from "./slug";
import type { FeedItem, FetchResult } from "./types";

export const FEED_X_URL = "https://raw.githubusercontent.com/zarazhangrui/follow-builders/main/feed-x.json";
export const FEED_PODCASTS_URL = "https://raw.githubusercontent.com/zarazhangrui/follow-builders/main/feed-podcasts.json";
export const FEED_BLOGS_URL = "https://raw.githubusercontent.com/zarazhangrui/follow-builders/main/feed-blogs.json";

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null;
}

function stringField(record: Record<string, unknown>, keys: string[]): string {
  for (const key of keys) {
    const value = record[key];
    if (typeof value === "string" && value.trim()) {
      return value.trim();
    }
  }
  return "";
}

function arrayField(record: Record<string, unknown>, key: string): unknown[] {
  const value = record[key];
  return Array.isArray(value) ? value : [];
}

function titleWithAuthor(author: string, title: string): string {
  return author ? `${author}: ${title}` : title;
}

export function parseXFeed(raw: unknown): FeedItem[] {
  if (!isRecord(raw)) return [];

  return arrayField(raw, "x").flatMap((builder) => {
    if (!isRecord(builder)) return [];

    const author = stringField(builder, ["name"]);
    const handle = stringField(builder, ["handle"]);
    const bio = stringField(builder, ["bio"]);

    return arrayField(builder, "tweets").flatMap((tweet) => {
      if (!isRecord(tweet)) return [];

      const tweetId = stringField(tweet, ["id"]);
      const body = stringField(tweet, ["text"]);
      const createdAt = stringField(tweet, ["createdAt", "publishedAt", "date"]);
      const url = stringField(tweet, ["url"]);

      if (!tweetId || !body || !createdAt || !url) return [];

      return [{
        id: `x:${tweetId}`,
        source: "x" as const,
        title: titleWithAuthor(author, truncateText(body, 80)),
        author,
        handle,
        bio,
        url,
        createdAt,
        body,
        metadata: {
          likes: tweet.likes as number | undefined,
          retweets: tweet.retweets as number | undefined,
          replies: tweet.replies as number | undefined,
          isQuote: tweet.isQuote as boolean | undefined,
          quotedTweetId: tweet.quotedTweetId as string | null | undefined
        }
      }];
    });
  });
}

export function parsePodcastFeed(raw: unknown): FeedItem[] {
  if (!isRecord(raw)) return [];

  return arrayField(raw, "podcasts").flatMap((podcast) => {
    if (!isRecord(podcast)) return [];

    const author = stringField(podcast, ["name", "podcast", "source", "show"]);
    const episodeTitle = stringField(podcast, ["title", "episodeTitle"]);
    const url = stringField(podcast, ["url", "link"]);
    const createdAt = stringField(podcast, ["publishedAt", "createdAt", "date"]);
    const body = stringField(podcast, ["transcript", "summary", "description", "content"]);

    if (!episodeTitle || !url || !createdAt || !body) return [];

    return [{
      id: `podcast:${url}`,
      source: "podcast" as const,
      title: titleWithAuthor(author, episodeTitle),
      author,
      url,
      createdAt,
      body,
      metadata: {
        duration: podcast.duration as string | number | undefined
      }
    }];
  });
}

export function parseBlogFeed(raw: unknown): FeedItem[] {
  if (!isRecord(raw)) return [];

  return arrayField(raw, "blogs").flatMap((blog) => {
    if (!isRecord(blog)) return [];

    const author = stringField(blog, ["source", "name", "site", "author"]);
    const postTitle = stringField(blog, ["title"]);
    const url = stringField(blog, ["url", "link"]);
    const createdAt = stringField(blog, ["publishedAt", "createdAt", "date"]);
    const body = stringField(blog, ["content", "text", "summary", "description"]);

    if (!postTitle || !url || !createdAt || !body) return [];

    return [{
      id: `blog:${url}`,
      source: "blog" as const,
      title: titleWithAuthor(author, postTitle),
      author,
      url,
      createdAt,
      body,
      metadata: {}
    }];
  });
}

export async function fetchJson(url: string): Promise<unknown> {
  const response = await requestUrl({ url, method: "GET" });
  return response.json;
}

export async function fetchEnabledFeeds(options: {
  syncX: boolean;
  syncPodcasts: boolean;
  syncBlogs: boolean;
}): Promise<FetchResult> {
  const items: FeedItem[] = [];
  const errors: string[] = [];

  const jobs: Array<Promise<void>> = [];

  if (options.syncX) {
    jobs.push(fetchJson(FEED_X_URL).then((raw) => {
      items.push(...parseXFeed(raw));
    }).catch((error: unknown) => {
      errors.push(`X feed failed: ${error instanceof Error ? error.message : String(error)}`);
    }));
  }

    jobs.push(fetchJson(FEED_BLOGS_URL).then((raw) => {
      items.push(...parseBlogFeed(raw));
    }).catch((error: unknown) => {
      errors.push(`Blog feed failed: ${error instanceof Error ? error.message : String(error)}`);
    }));
```

```ts
import { normalizePath, TFile, TFolder, Vault } from "obsidian";
import { renderFeedItemMarkdown } from "./markdown";
import { dateFolderFromIso, safeFileName } from "./slug";
import type { FeedItem } from "./types";

type MinimalFile = { path: string };

export interface MinimalVault {
  getAbstractFileByPath(path: string): unknown;
  createFolder(path: string): Promise<unknown>;
  create(path: string, data: string): Promise<unknown>;
  modify(file: MinimalFile, data: string): Promise<void>;
}

export interface WriteOptions {
  overwriteExisting: boolean;
  syncedAt: string;
}

export interface WriteResult {
  status: "created" | "updated" | "skipped";
  path: string;
}

function itemIdSuffix(item: FeedItem): string {
  const suffix = item.id.split(":").pop() ?? item.id;
  return safeFileName(suffix, 40);
}

export function buildItemPath(rootFolder: string, item: FeedItem): string {
  const dateFolder = dateFolderFromIso(item.createdAt);
  const titleSlug = safeFileName(item.title, 64);
  return normalizePath(`${rootFolder}/${dateFolder}/${item.source}-${titleSlug}-${itemIdSuffix(item)}.md`);
}

function isExistingFolder(value: unknown): boolean {
  if (value instanceof TFolder) return true;
  return Boolean(value && typeof value === "object" && "kind" in value && (value as { kind: unknown }).kind === "folder");
}

function isExistingFile(value: unknown): value is MinimalFile {
  if (value instanceof TFile) return true;
  return Boolean(value && typeof value === "object" && "path" in value && (value as { path: unknown }).path);
}

async function ensureFolder(vault: MinimalVault, folderPath: string): Promise<void> {
  const normalized = normalizePath(folderPath);
  const existing = vault.getAbstractFileByPath(normalized);

  if (isExistingFolder(existing)) return;
  if (isExistingFile(existing)) {
    throw new Error(`Cannot create folder because a file already exists at ${normalized}`);
  }

  const parent = normalized.split("/").slice(0, -1).join("/");
  if (parent) {
    await ensureFolder(vault, parent);
  }

  await vault.createFolder(normalized);
}

export async function writeFeedItem(
  vault: MinimalVault | Vault,
  rootFolder: string,
  item: FeedItem,
  options: WriteOptions
): Promise<WriteResult> {
  const path = buildItemPath(rootFolder, item);
  const folder = path.split("/").slice(0, -1).join("/");
  await ensureFolder(vault, folder);

  const markdown = renderFeedItemMarkdown(item, options.syncedAt);
  const existing = vault.getAbstractFileByPath(path);

  if (isExistingFile(existing)) {
    if (!options.overwriteExisting) {
      return { status: "skipped", path };
    }

    await vault.modify(existing, markdown);
    return { status: "updated", path };
  }

  await vault.create(path, markdown);
  return { status: "created", path };
}
```

```ts
import { fetchEnabledFeeds } from "./feeds";
import { writeFeedItem, type WriteResult } from "./writer";
import type { FeedItem, FetchResult, FollowBuildersSettings, FollowBuildersSyncState, SyncResult } from "./types";

export interface RunSyncDependencies {
  settings: FollowBuildersSettings;
  state: FollowBuildersSyncState;
  fetchFeeds?: (options: FollowBuildersSettings) => Promise<FetchResult>;
  writeItem: (item: FeedItem, syncedAt: string) => Promise<WriteResult>;
  now?: () => Date;
}

export async function runSync(deps: RunSyncDependencies): Promise<SyncResult> {
  const fetchFeeds = deps.fetchFeeds ?? fetchEnabledFeeds;
  const now = deps.now ?? (() => new Date());
  const syncedAt = now().toISOString();
  const result: SyncResult = {
    created: 0,
    updated: 0,
    skipped: 0,
    failed: 0,
    errors: []
  };

  const fetched = await fetchFeeds(deps.settings);
  result.errors.push(...fetched.errors);

  for (const item of fetched.items) {
    if (deps.state.syncedIds[item.id] && !deps.settings.overwriteExisting) {
      result.skipped += 1;
      continue;
    }

    try {
      const writeResult = await deps.writeItem(item, syncedAt);

      if (writeResult.status === "created") result.created += 1;
      if (writeResult.status === "updated") result.updated += 1;
      if (writeResult.status === "skipped") result.skipped += 1;
      deps.state.syncedIds[item.id] = true;
    } catch (error) {
      result.failed += 1;
      result.errors.push(`Failed to write ${item.id}: ${error instanceof Error ? error.message : String(error)}`);
    }
  }

  deps.state.lastSyncedAt = syncedAt;
  return result;
}
```

```ts
import { FollowBuildersSettingTab } from "./settings";
import { runSync } from "./sync";
import { DEFAULT_SETTINGS, createDefaultState, type FollowBuildersSettings, type FollowBuildersSyncState } from "./types";
import { writeFeedItem } from "./writer";

export default class FollowBuildersSyncPlugin extends Plugin {
  settings: FollowBuildersSettings = { ...DEFAULT_SETTINGS };
  state: FollowBuildersSyncState = createDefaultState();
  private syncing = false;

  async onload(): Promise<void> {
    await this.loadPluginData();

    this.addRibbonIcon("refresh-cw", "Sync Follow Builders", () => {
      void this.syncFeeds();
    });

    this.addCommand({
      id: "sync-follow-builders-feeds",
      name: "Sync Follow Builders feeds",
      callback: () => {
        void this.syncFeeds();
      }
    });

    this.addSettingTab(new FollowBuildersSettingTab(this.app, this));
  }

  async loadPluginData(): Promise<void> {
    const data = await this.loadData();
    this.settings = {
      ...DEFAULT_SETTINGS,
      ...(data?.settings ?? data ?? {})
    };
    this.state = {
      ...createDefaultState(),
      ...(data?.state ?? {})
    };
  }

  async savePluginData(): Promise<void> {
    await this.saveData({
      settings: this.settings,
      state: this.state
    });
  }

  async clearSyncHistory(): Promise<void> {
    this.state = createDefaultState();
    await this.savePluginData();
    new Notice("Follow Builders sync history cleared.");
  }

  async syncFeeds(): Promise<void> {
    if (this.syncing) {
      new Notice("Follow Builders sync is already running.");
      return;
    }

    this.syncing = true;
    new Notice("Syncing Follow Builders feeds...");

    try {
      const result = await runSync({
        settings: this.settings,
        state: this.state,
        writeItem: (item, syncedAt) => writeFeedItem(this.app.vault, this.settings.targetFolder, item, {
          overwriteExisting: this.settings.overwriteExisting,
          syncedAt
        })
      });

      await this.savePluginData();

      const problems = result.failed + result.errors.length;
      new Notice(
        `Follow Builders sync complete: ${result.created} created, ${result.updated} updated, ${result.skipped} skipped${problems ? `, ${problems} issues` : ""}.`,
        8000
      );
    } catch (error) {
      new Notice(`Follow Builders sync failed: ${error instanceof Error ? error.message : String(error)}`, 10000);
    } finally {
      this.syncing = false;
    }
  }
}
```
